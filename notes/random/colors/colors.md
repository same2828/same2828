# Normal colors (0-7)
palette = 0=#21222C   # black
palette = 1=#FF5555   # red
palette = 2=#50FA7B   # green
palette = 3=#F1FA8C   # yellow
palette = 4=#BD93F9   # blue/purple #be93f9
palette = 5=#FF79C6   # purple
palette = 6=#8BE9FD   # aqua
palette = 7=#F8F8F2   # white
# Bright colors (8-15)

palette = 9=#FF6E6E   # bright red
palette = 10=#69FF94  # bright green
palette = 11=#FFFFA5  # bright yellow
palette = 12=#D6ACFF  # bright blue
palette = 13=#FF92DF  # bright purple
palette = 14=#A4FFFF  # bright aqua
palette = 15=#FFFFFF  # bright white
background = #282A36
foreground = #F8F8F2
cursor-color = #F8F8F2
cursor-text = #282A36
selection-background = #44475A
selection-foreground = #FFFFFF
